The assembly fo Culex pipiens pipiens (accession number GCA_963924435.1) was downloaded from NCBI genome database. The seqID were replaced by new seqIDs containing the number of the chromosomes with:
```perl 
perl -ne 'chomp($_); if ($_ =~ m/>(.+?)\s+.+chromosome: (\d+)/) { $oid=$1; $id="CupipDToLv1C".$2; print "$oid\t$id\n"} elsif ($_ =~ m/>(.+?)\s+.+SUPER_1_unloc_1/) {$oid=$1; $id="CupipDToLv1C1UCTG1"; print "$oid\t$id\n"} else { if ($_ =~ m/>(.+?)\s+.+scaffold_(\d+)/) {$oid=$1; $id="CupipDToLv1USCF".sprintf("%03s", $2); print "$oid\t$id\n" } elsif ($_ =~ m/>(.+?)\s+.+mitochondrion/) {$oid=$1; $id="CupipDToLv1MIT"; print "$oid\t$id\n"} else { }}' GCA_963924435.1_idCulPipi1.1_genomic.fna > Culex_pipiens_SAN22000250v1_GCA963924435_genome.equiv_seqID.txt


perl -ne 'chomp($_); if ($_ =~ m/>(.+?)\s+.+chromosome: (\d+)/) { $oid=$1; $id="CupipDToLv1C".$2; print ">$id\n"} elsif ($_ =~ m/>(.+?)\s+.+SUPER_1_unloc_1/) {$oid=$1; $id="CupipDToLv1C1UCTG1"; print ">$id\n"} else { if ($_ =~ m/>(.+?)\s+.+scaffold_(\d+)/) {$oid=$1; $id="CupipDToLv1USCF".sprintf("%03s", $2); print ">$id\n" } elsif ($_ =~ m/>(.+?)\s+.+mitochondrion/) {$oid=$1; $id="CupipDToLv1MIT"; print ">$id\n"} else { print "$_\n"; }}' GCA_963924435.1_idCulPipi1.1_genomic.fna > Culex_pipiens_SAN22000250v1_GCA963924435_genome.fasta
```
